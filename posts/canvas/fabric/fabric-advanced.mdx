# Fabric.js 进阶指南

在掌握了 Fabric.js 的基础用法后，我们来探索它的高级特性。这些功能将帮助你构建真正的生产级应用，如在线设计工具、图片编辑器、白板协作系统等。

## 1. 自定义控制点 (Custom Controls)

默认情况下，Fabric 对象有 8 个控制点（4 个角 + 4 个边）。但你可以完全自定义这些控制点的行为、外观和位置。

### 自定义控制点样式

<CodePenDemo
  title="自定义控制点样式"
  height="400px"
  code={`const CustomControls = () => {
  const canvasEl = React.useRef(null);

  React.useEffect(() => {
    const canvas = new fabric.Canvas(canvasEl.current, {
      width: 600,
      height: 350,
      backgroundColor: '#f8f9fa'
    });

    // 创建一个矩形
    const rect = new fabric.Rect({
      left: 150,
      top: 100,
      width: 200,
      height: 150,
      fill: '#6366f1',
      cornerColor: '#ec4899',
      cornerSize: 12,
      transparentCorners: false,
      cornerStyle: 'circle', // 圆形控制点
      borderColor: '#8b5cf6',
      borderScaleFactor: 2,
      padding: 10
    });

    // 自定义旋转图标
    rect.setControlsVisibility({
      mt: false, // 隐藏顶部中间控制点
      mb: false, // 隐藏底部中间控制点
      ml: false, // 隐藏左侧中间控制点
      mr: false  // 隐藏右侧中间控制点
    });

    canvas.add(rect);
    canvas.setActiveObject(rect);

    // 创建一个圆形，使用不同的控制点样式
    const circle = new fabric.Circle({
      left: 400,
      top: 100,
      radius: 60,
      fill: '#10b981',
      cornerColor: '#f59e0b',
      cornerSize: 16,
      cornerStrokeColor: '#dc2626',
      borderColor: '#3b82f6',
      transparentCorners: false
    });

    canvas.add(circle);

    return () => canvas.dispose();
  }, []);

  return (
    <div className="flex justify-center p-4">
      <canvas ref={canvasEl} />
    </div>
  );
}

render(<CustomControls />)`}

      x: 0.5,
      y: -0.5,
      offsetY: -16,
      offsetX: 16,
      cursorStyle: 'pointer',
      mouseUpHandler: function(eventData, transformData) {
        const target = transformData.target;
        const canvas = target.canvas;
        canvas.remove(target);
        canvas.requestRenderAll();
        return true;
      },
      render: function(ctx, left, top, styleOverride, fabricObject) {
        const size = this.cornerSize || 24;
        ctx.save();
        ctx.translate(left, top);
        ctx.rotate(fabric.util.degreesToRadians(fabricObject.angle || 0));
        
        // 绘制红色圆形背景
        ctx.fillStyle = '#F44336';
        ctx.beginPath();
        ctx.arc(0, 0, size/2, 0, 2 * Math.PI);
        ctx.fill();
        
        // 绘制白色 X
        ctx.strokeStyle = 'white';
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.moveTo(-size/4, -size/4);
        ctx.lineTo(size/4, size/4);
        ctx.moveTo(size/4, -size/4);
        ctx.lineTo(-size/4, size/4);
        ctx.stroke();
        
        ctx.restore();
      },
      cornerSize: 24
    });

    // 为每个对象添加删除控制点
    rect.controls.deleteControl = deleteControl;
    circle.controls.deleteControl = deleteControl;

    canvas.add(rect, circle);
    canvas.setActiveObject(rect);

    return () => canvas.dispose();
  }, []);

  return (
    <div className="flex flex-col items-center gap-2">
      <canvas ref={canvasEl} />
      <p className="text-sm text-gray-600">点击对象右上角的红色 × 图标删除对象</p>

## 2. 图像滤镜 (Image Filters)

Fabric.js 内置了丰富的图像滤镜，可以对图片进行各种处理。

常用滤镜：
- `Grayscale`: 灰度
- `Sepia`: 褐色调
- `Brightness`: 亮度
- `Contrast`: 对比度
- `Blur`: 模糊
- `Pixelate`: 像素化
- `RemoveColor`: 移除颜色

<CodePenDemo
  title="图像滤镜效果"
  height="500px"
  code={`const ImageFilters = () => {
  const canvasEl = React.useRef(null);
  const canvasRef = React.useRef(null);
  const imageRef = React.useRef(null);
  const [currentFilter, setCurrentFilter] = React.useState('none');

  React.useEffect(() => {
    const canvas = new fabric.Canvas(canvasEl.current, {
      width: 600,
      height: 400,
      backgroundColor: '#f0f0f0'
    });
    canvasRef.current = canvas;

    // 创建一个彩色图像源
    const imgCanvas = document.createElement('canvas');
    imgCanvas.width = 300;
    imgCanvas.height = 200;
    const ctx = imgCanvas.getContext('2d');
    
    // 绘制渐变
    const gradient = ctx.createLinearGradient(0, 0, 300, 0);
    gradient.addColorStop(0, '#ff6b6b');
    gradient.addColorStop(0.33, '#4ecdc4');
    gradient.addColorStop(0.66, '#45b7d1');
    gradient.addColorStop(1, '#f7b731');
    ctx.fillStyle = gradient;
    ctx.fillRect(0, 0, 300, 200);
    
    // 添加一些图案
    ctx.fillStyle = 'rgba(255, 255, 255, 0.3)';
    for (let i = 0; i < 10; i++) {
      ctx.beginPath();
      ctx.arc(
        Math.random() * 300,
        Math.random() * 200,
        Math.random() * 30 + 10,
        0,
        Math.PI * 2
      );
      ctx.fill();
    }

    // 从 canvas 创建 fabric.Image
    fabric.Image.fromObject({
      src: imgCanvas.toDataURL(),
      left: 150,
      top: 100,
      selectable: false
    }).then((img) => {
      imageRef.current = img;
      canvas.add(img);
      canvas.requestRenderAll();
    });

    return () => canvas.dispose();
  }, []);

  const applyFilter = (filterType) => {
    const canvas = canvasRef.current;
    const obj = imageRef.current;
    
    if (!obj) return;

    // 清除所有滤镜
    obj.filters = [];

    // Fabric.js v7 使用 fabric.filters
    switch(filterType) {
      case 'grayscale':
        obj.filters.push(new fabric.filters.Grayscale());
        break;
      case 'sepia':
        obj.filters.push(new fabric.filters.Sepia());
        break;
      case 'brightness':
        obj.filters.push(new fabric.filters.Brightness({ brightness: 0.3 }));
        break;
      case 'contrast':
        obj.filters.push(new fabric.filters.Contrast({ contrast: 0.5 }));
        break;
      case 'blur':
        obj.filters.push(new fabric.filters.Blur({ blur: 0.5 }));
        break;
      case 'pixelate':
        obj.filters.push(new fabric.filters.Pixelate({ blocksize: 8 }));
        break;
      case 'invert':
        obj.filters.push(new fabric.filters.Invert());
        break;
    }

    obj.applyFilters();
    canvas.requestRenderAll();
    setCurrentFilter(filterType);
  };

  return (
    <div className="flex flex-col items-center gap-4">
      <canvas ref={canvasEl} />
      <div className="flex flex-wrap gap-2 max-w-[600px] justify-center">

                ? 'bg-blue-600 text-white' 
                : 'bg-gray-200 hover:bg-gray-300 text-gray-800'
            }\`}
          >
            {filter}
          </button>
        ))}
      </div>
    </div>
  );
}

## 3. 自由绘制模式 (Free Drawing)

Fabric 支持自由绘制模式，用户可以用鼠标或触摸屏直接在画布上绘画。这对于白板、签名、标注等场景非常有用。

<CodePenDemo
  title="自由绘制白板"
  height="500px"
  code={`const FreeDrawing = () => {
  const canvasEl = React.useRef(null);
  const canvasRef = React.useRef(null);
  const [isDrawing, setIsDrawing] = React.useState(false);
  const [brushColor, setBrushColor] = React.useState('#000000');
  const [brushWidth, setBrushWidth] = React.useState(5);

  React.useEffect(() => {
    const canvas = new fabric.Canvas(canvasEl.current, {
      width: 600,
      height: 400,
      backgroundColor: '#ffffff',
      isDrawingMode: false
    });
    canvasRef.current = canvas;

    // 配置画笔
    canvas.freeDrawingBrush = new fabric.PencilBrush(canvas);
    canvas.freeDrawingBrush.color = brushColor;
    canvas.freeDrawingBrush.width = brushWidth;

    return () => canvas.dispose();
  }, []);

  React.useEffect(() => {
    const canvas = canvasRef.current;
    if (canvas) {
      canvas.freeDrawingBrush.color = brushColor;
    }
  }, [brushColor]);

  React.useEffect(() => {
    const canvas = canvasRef.current;
    if (canvas) {
      canvas.freeDrawingBrush.width = brushWidth;
    }
  }, [brushWidth]);

  const toggleDrawing = () => {
    const canvas = canvasRef.current;
    const newMode = !isDrawing;
    canvas.isDrawingMode = newMode;
    setIsDrawing(newMode);
  };

  const clearCanvas = () => {
    const canvas = canvasRef.current;
    canvas.clear();
    canvas.backgroundColor = '#ffffff';
    canvas.requestRenderAll();
  };

  const undo = () => {
    const canvas = canvasRef.current;
    const objects = canvas.getObjects();
    if (objects.length > 0) {
      canvas.remove(objects[objects.length - 1]);
    }
  };

  return (
    <div className="flex flex-col items-center gap-4">
      <div className="border-2 border-gray-300 rounded">
        <canvas ref={canvasEl} />
      </div>
      
      <div className="flex flex-wrap gap-3 items-center justify-center max-w-[600px]">
        <button
          onClick={toggleDrawing}
          className={\`px-4 py-2 rounded font-medium transition \${
            isDrawing 
              ? 'bg-red-500 hover:bg-red-600 text-white' 
              : 'bg-green-500 hover:bg-green-600 text-white'
          }\`}
        >
          {isDrawing ? '停止绘制' : '开始绘制'}
        </button>

        <div className="flex items-center gap-2">
          <label className="text-sm font-medium">颜色:</label>
          <input
            type="color"
            value={brushColor}
            onChange={(e) => setBrushColor(e.target.value)}
            className="w-12 h-10 rounded cursor-pointer"
          />
        </div>

        <div className="flex items-center gap-2">
          <label className="text-sm font-medium">粗细:</label>
          <input
            type="range"
            min="1"
            max="50"
            value={brushWidth}
            onChange={(e) => setBrushWidth(Number(e.target.value))}
            className="w-32"
          />
          <span className="text-sm w-8">{brushWidth}</span>
        </div>

        <button
          onClick={undo}
          className="px-4 py-2 bg-gray-500 hover:bg-gray-600 text-white rounded transition"
        >
          撤销
        </button>

        <button
          onClick={clearCanvas}
          className="px-4 py-2 bg-orange-500 hover:bg-orange-600 text-white rounded transition"
        >
          清空
        </button>
      </div>
    </div>
  );
}

render(<FreeDrawing />)`}

## 6. 性能优化技巧

在处理大量对象或复杂场景时，性能优化至关重要：

### 技巧 1: 对象缓存

```javascript
// 对复杂对象启用缓存
object.set({ objectCaching: true });

// 当对象不再变化时，缓存可以显著提升渲染性能
```

### 技巧 2: 批量操作

```javascript
// ❌ 不好的做法
for (let i = 0; i < 100; i++) {
  canvas.add(new fabric.Rect({...}));
  canvas.requestRenderAll(); // 每次都重绘
}

// ✅ 好的做法
const objects = [];
for (let i = 0; i < 100; i++) {
  objects.push(new fabric.Rect({...}));
}
canvas.add(...objects);
canvas.requestRenderAll(); // 只重绘一次
```

### 技巧 3: 禁用不需要的功能

```javascript
// 如果不需要选择功能
canvas.selection = false;

// 禁用对象的某些控制点
object.setControlsVisibility({
  mt: false,
  mb: false,
  ml: false,
  mr: false
});
```

### 技巧 4: 使用 StaticCanvas

如果画布内容不需要交互，使用 `StaticCanvas` 代替 `Canvas`：

```javascript
const canvas = new fabric.StaticCanvas('canvas');
// StaticCanvas 没有事件监听，性能更好
```

## 7. 实战：简易图片编辑器

让我们综合运用这些知识，创建一个功能完整的小型图片编辑器：

<CodePenDemo
  title="高级图片编辑器 Pro"
  height="700px"
  code={`const ImageEditor = () => {
  const canvasEl = React.useRef(null);
  const canvasRef = React.useRef(null);
  const [selectedObj, setSelectedObj] = React.useState(null);
  const [attributes, setAttributes] = React.useState({ fill: '#3b82f6', opacity: 1, strokeWidth: 0 });

    canvas.on('selection:created', (e) => updateSelection(e.selected[0]));

        if (active && !active.isEditing) canvas.remove(active);
      }

    window.addEventListener('keydown', handleKeyDown);

    return () => {
      canvas.dispose();
      window.removeEventListener('keydown', handleKeyDown);
    };
  }, []);

        img.scaleToWidth(200);
        canvasRef.current.add(img).centerObject(img).setActiveObject(img);
      });
    };
    reader.readAsDataURL(file);
  };

  const changeStack = (dir) => {

    if (dir === 'front') obj.bringToFront();
    if (dir === 'back') obj.sendToBack();
    canvasRef.current.requestRenderAll();
  };

  return (
    <div className="flex bg-gray-50 rounded-lg overflow-hidden border border-gray-200">
      {/* 工具栏 */}
      <div className="w-16 bg-white border-r flex flex-col items-center py-4 gap-4 shadow-sm">
        <button onClick={() => {
          const r = new fabric.Rect({ left: 50, top: 50, width: 100, height: 100, fill: '#3b82f6' });
          canvasRef.current.add(r).setActiveObject(r);
        }} title="矩形" className="p-2 hover:bg-blue-50 rounded text-xl">🟦</button>
        <button onClick={() => {
          const c = new fabric.Circle({ left: 50, top: 50, radius: 50, fill: '#10b981' });
          canvasRef.current.add(c).setActiveObject(c);
        }} title="圆形" className="p-2 hover:bg-green-50 rounded text-xl">🟢</button>
        <button onClick={() => {
          const t = new fabric.IText('新文字', { left: 50, top: 50, fontSize: 20 });
          canvasRef.current.add(t).setActiveObject(t);
        }} title="文字" className="p-2 hover:bg-yellow-50 rounded text-xl">✍️</button>
        <label className="p-2 hover:bg-purple-50 rounded text-xl cursor-pointer">
          🖼️<input type="file" hidden onChange={addImage} accept="image/*" />
        </label>
      </div>

      {/* 画布区域 */}
      <div className="flex-1 p-6 flex justify-center bg-gray-200">
        <div className="shadow-2xl bg-white leading-[0]">
          <canvas ref={canvasEl} />
        </div>
      </div>

      {/* 属性面板 */}
      <div className="w-64 bg-white border-l p-4 flex flex-col gap-6">
        <h3 className="font-bold text-gray-700 border-b pb-2">属性编辑器</h3>
        
        {selectedObj ? (
          <div className="flex flex-col gap-4">
            <div className="space-y-2">
              <label className="text-xs text-gray-500 uppercase font-semibold">填充颜色</label>
              <input 
                type="color" 
                className="w-full h-8 rounded border"
                value={attributes.fill} 
                onChange={(e) => handleAttrChange('fill', e.target.value)} 
              />
            </div>

            <div className="space-y-2">
              <label className="text-xs text-gray-500 uppercase font-semibold">不透明度 ({Math.round(attributes.opacity * 100)}%)</label>
              <input 
                type="range" min="0" max="1" step="0.1" className="w-full"
                value={attributes.opacity} 
                onChange={(e) => handleAttrChange('opacity', e.target.value)} 
              />
            </div>

            <div className="space-y-2">
              <label className="text-xs text-gray-500 uppercase font-semibold">层级管理</label>
              <div className="flex gap-2">
                <button onClick={() => changeStack('front')} className="flex-1 py-1 bg-gray-100 hover:bg-gray-200 rounded text-xs">置于顶层</button>
                <button onClick={() => changeStack('back')} className="flex-1 py-1 bg-gray-100 hover:bg-gray-200 rounded text-xs">置于底层</button>
              </div>
            </div>

            <button 
              onClick={() => canvasRef.current.remove(selectedObj)}
              className="mt-4 w-full py-2 bg-red-50 text-red-600 hover:bg-red-100 rounded text-sm font-medium transition-colors"
            >
              删除选中元素
            </button>
          </div>
        ) : (
          <div className="text-center py-20 text-gray-400 italic text-sm">
            请选中画布上的元素进行编辑
          </div>
        )}

        <div className="mt-auto border-t pt-4 space-y-2">
           <button onClick={() => {
             const data = canvasRef.current.toDataURL({ format: 'png' });
             const link = document.createElement('a');
             link.download = 'export.png';
             link.href = data;
             link.click();
           }} className="w-full py-2 bg-indigo-600 text-white rounded hover:bg-indigo-700 text-sm font-medium shadow-sm">
             🚀 导出为 PNG
           </button>
           <button onClick={() => canvasRef.current.clear().set({backgroundColor: '#f8fafc'})} className="w-full py-2 text-gray-500 hover:text-gray-700 text-sm">
             清空画布
           </button>
        </div>
      </div>
    </div>
  );
}
