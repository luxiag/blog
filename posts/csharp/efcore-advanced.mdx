public class ProductSearchCriteria
{
    public string Name { get; set; }
    public decimal? MinPrice { get; set; }
    public decimal? MaxPrice { get; set; }
    public string Category { get; set; }
    public bool? InStock { get; set; }

public async Task<List<Product>> SearchProducts(ProductSearchCriteria criteria)
{
    var query = context.Products.AsQueryable();
    
    // 动态添加条件
    if (!string.IsNullOrEmpty(criteria.Name))
    {
        query = query.Where(p => p.Name.Contains(criteria.Name));
    }
    
    if (criteria.MinPrice.HasValue)
    {
        query = query.Where(p => p.Price >= criteria.MinPrice.Value);
    }
    
    if (criteria.MaxPrice.HasValue)
    {
        query = query.Where(p => p.Price <= criteria.MaxPrice.Value);
    }
    
    if (!string.IsNullOrEmpty(criteria.Category))
    {
        query = query.Where(p => p.Category == criteria.Category);
    }
    
    if (criteria.InStock.HasValue)
    {
        query = query.Where(p => p.Stock > 0 == criteria.InStock.Value);
    }
    
    return await query.ToListAsync();
}

#### 使用表达式树构建动态查询

```csharp
using System.Linq.Expressions;

public class ExpressionBuilder
{
    // 构建 AND 条件
    public static Expression<Func<T, bool>> And<T>(
        Expression<Func<T, bool>> expr1,
        Expression<Func<T, bool>> expr2)
    {
        var parameter = Expression.Parameter(typeof(T));
        
        var leftVisitor = new ReplaceExpressionVisitor(expr1.Parameters[0], parameter);
        var left = leftVisitor.Visit(expr1.Body);
        
        var rightVisitor = new ReplaceExpressionVisitor(expr2.Parameters[0], parameter);
        var right = rightVisitor.Visit(expr2.Body);
        
        return Expression.Lambda<Func<T, bool>>(
            Expression.AndAlso(left, right), parameter);
    }
    
    // 构建 OR 条件
    public static Expression<Func<T, bool>> Or<T>(
        Expression<Func<T, bool>> expr1,
        Expression<Func<T, bool>> expr2)
    {
        var parameter = Expression.Parameter(typeof(T));
        
        var leftVisitor = new ReplaceExpressionVisitor(expr1.Parameters[0], parameter);
        var left = leftVisitor.Visit(expr1.Body);
        
        var rightVisitor = new ReplaceExpressionVisitor(expr2.Parameters[0], parameter);
        var right = rightVisitor.Visit(expr2.Body);
        
        return Expression.Lambda<Func<T, bool>>(
            Expression.OrElse(left, right), parameter);
    }
}

class ReplaceExpressionVisitor : ExpressionVisitor
{
    private readonly Expression oldValue;
    private readonly Expression newValue;
    
    public ReplaceExpressionVisitor(Expression oldValue, Expression newValue)
    {
        this.oldValue = oldValue;
        this.newValue = newValue;
    }
    
    public override Expression Visit(Expression node)
    {
        return node == oldValue ? newValue : base.Visit(node);
    }
}

// 使用
Expression<Func<Product, bool>> nameFilter = p => p.Name.Contains("手机");
Expression<Func<Product, bool>> priceFilter = p => p.Price > 1000;

var combinedFilter = ExpressionBuilder.And(nameFilter, priceFilter);
var products = await context.Products.Where(combinedFilter).ToListAsync();
```

### 6. 递归查询（层级数据）

```csharp
// 实体定义
public class Category
{
    public int Id { get; set; }
    public string Name { get; set; }
    public int? ParentId { get; set; }
    
    public Category Parent { get; set; }
    public List<Category> Children { get; set; }
}

// 获取所有子类别（递归）
public async Task<List<Category>> GetAllDescendants(int categoryId)
{
    var descendants = new List<Category>();
    var queue = new Queue<int>();
    queue.Enqueue(categoryId);
    
    while (queue.Count > 0)
    {
        var currentId = queue.Dequeue();
        var children = await context.Categories
            .Where(c => c.ParentId == currentId)
            .ToListAsync();
            
        descendants.AddRange(children);
        
        foreach (var child in children)
        {
            queue.Enqueue(child.Id);
        }
    }
    
    return descendants;
}

// 使用原始 SQL（更高效）
var descendants = await context.Categories
    .FromSqlRaw(@"
        WITH RECURSIVE CategoryTree AS (
            SELECT * FROM Categories WHERE Id = {0}
            UNION ALL
            SELECT c.* FROM Categories c
            INNER JOIN CategoryTree ct ON c.ParentId = ct.Id
        )
        SELECT * FROM CategoryTree WHERE Id != {0}", categoryId)
    .ToListAsync();
```

### 2. 乐观并发控制（Optimistic Concurrency）

#### 使用 RowVersion

```csharp
public class Product
{
    public int Id { get; set; }
    public string Name { get; set; }
    public int Stock { get; set; }
    
    // 并发令牌
    [Timestamp]
    public byte[] RowVersion { get; set; }
}

// 配置
modelBuilder.Entity<Product>()
    .Property(p => p.RowVersion)
    .IsRowVersion();

// 使用
try
{
    var product = await context.Products.FindAsync(1);
    product.Stock -= 10;
    await context.SaveChangesAsync();
}
catch (DbUpdateConcurrencyException ex)
{
    // 并发冲突处理
    var entry = ex.Entries.Single();
    var databaseValues = await entry.GetDatabaseValuesAsync();
    
    if (databaseValues == null)
    {
        // 记录已被删除
        Console.WriteLine("产品已被删除");
    }
    else
    {
        // 记录已被修改
        var databaseProduct = (Product)databaseValues.ToObject();
        Console.WriteLine($"数据库中的库存：{databaseProduct.Stock}");
        Console.WriteLine($"你的修改：{((Product)entry.Entity).Stock}");
        
        // 选择处理策略
        // 1. 使用数据库值
        entry.OriginalValues.SetValues(databaseValues);
        
        // 2. 使用客户端值（覆盖）
        // entry.OriginalValues.SetValues(entry.GetDatabaseValues());
        
        // 3. 合并值
        var clientProduct = (Product)entry.Entity;
        clientProduct.Stock = databaseProduct.Stock - 10;
        entry.OriginalValues.SetValues(databaseValues);
        
        // 重试保存
        await context.SaveChangesAsync();
    }
}
```

### 4. 并发控制最佳实践

```csharp
public class ConcurrencyHandler
{
    private readonly AppDbContext context;
    private readonly ILogger<ConcurrencyHandler> logger;
    
    public ConcurrencyHandler(AppDbContext context, ILogger<ConcurrencyHandler> logger)
    {
        this.context = context;
        this.logger = logger;
    }
    
    public async Task<bool> UpdateWithRetry<T>(
        T entity,
        Action<T> updateAction,
        int maxRetries = 3) where T : class
    {
        for (int i = 0; i < maxRetries; i++)
        {
            try
            {
                updateAction(entity);
                await context.SaveChangesAsync();
                return true;
            }
            catch (DbUpdateConcurrencyException ex)
            {
                logger.LogWarning($"并发冲突，重试 {i + 1}/{maxRetries}");
                
                if (i == maxRetries - 1)
                {
                    throw;
                }
                
                // 刷新实体
                var entry = ex.Entries.Single();
                await entry.ReloadAsync();
            }
        }
        
        return false;
    }
}

// 使用
await concurrencyHandler.UpdateWithRetry(
    product,
    p => p.Stock -= 10,
    maxRetries: 5
);
```

// 分页辅助类
public class PagedResult<T>
{
    public List<T> Items { get; set; }
    public int TotalCount { get; set; }
    public int PageNumber { get; set; }
    public int PageSize { get; set; }
    public int TotalPages => (int)Math.Ceiling(TotalCount / (double)PageSize);
}

public async Task<PagedResult<T>> GetPagedAsync<T>(
    IQueryable<T> query,
    int pageNumber,
    int pageSize)
{
    var totalCount = await query.CountAsync();
    
    var items = await query
        .Skip((pageNumber - 1) * pageSize)
        .Take(pageSize)
        .ToListAsync();
    
    return new PagedResult<T>
    {
        Items = items,
        TotalCount = totalCount,
        PageNumber = pageNumber,
        PageSize = pageSize
    };
}

### 10. 缓存策略

```csharp
public class CachedRepository<T> where T : class
{
    private readonly AppDbContext context;
    private readonly IMemoryCache cache;
    
    public CachedRepository(AppDbContext context, IMemoryCache cache)
    {
        this.context = context;
        this.cache = cache;
    }
    
    public async Task<T> GetByIdAsync(int id)
    {
        string cacheKey = $"{typeof(T).Name}_{id}";
        
        if (!cache.TryGetValue(cacheKey, out T entity))
        {
            entity = await context.Set<T>().FindAsync(id);
            
            if (entity != null)
            {
                cache.Set(cacheKey, entity, TimeSpan.FromMinutes(10));
            }
        }
        
        return entity;
    }
    
    public async Task UpdateAsync(T entity)
    {
        context.Set<T>().Update(entity);
        await context.SaveChangesAsync();
        
        // 清除缓存
        var id = context.Entry(entity).Property("Id").CurrentValue;
        string cacheKey = $"{typeof(T).Name}_{id}";
        cache.Remove(cacheKey);
    }
}
```

// ✅ 优化后
public async Task<OrderStatistics> GetOrderStatsOptimized(
    DateTime startDate, DateTime endDate)
{
    var stats = await context.Orders
        .Where(o => o.OrderDate >= startDate && o.OrderDate <= endDate)
        .GroupBy(o => 1)  // 分组技巧
        .Select(g => new OrderStatistics
        {
            TotalOrders = g.Count(),
            TotalRevenue = g.Sum(o => o.Total),
            AverageOrderValue = g.Average(o => o.Total),
            TotalItems = g.Sum(o => o.Items.Count)
        })
        .FirstOrDefaultAsync();
    
    return stats ?? new OrderStatistics();
}
